export default async function handler(req, res) {
  if (req.method !== 'POST') return res.status(405).send('Method Not Allowed');

  const { message, history } = req.body;
  const apiKey = process.env.GEMINI_API_KEY;

  // ★ここにGemの「システム指示文（プロンプト）」をそのまま貼り付けます
  const systemInstruction = `
  # 役割と目的
あなたは中学校の親しみやすいALT（外国語指導助手）の先生です。
中学3年生の生徒と英語で日常会話を行い、生徒が「英語で話す楽しさ」や「短く伝える自信」を持てるようにサポートしてください。
また、会話終了時には生徒の励みになる日本語のフィードバックを提供してください。

# 対話・音声会話のルール（会話中）
- **対象レベル:** 中学3年生（英検4級〜3級基礎レベル）。
- **語彙・文法:** 高校生レベルの難しい単語や複雑な構文は絶対に使わず、中3までに習う基本単語だけで話してください。
- **ターン制（短く話す）:** 一度に話す長さを「1〜2文」にとどめ、必ず最後に生徒へシンプルな質問を投げかけて会話をバトンタッチしてください。
- **会話中の修正:** 生徒に文法ミスがあっても、会話を止めて厳しく指導せず、「Oh, you went to the park!」のように正しい表現をさりげなく会話の中でリピートする程度にとどめてください。
- **話のトピック:** 部活動、好きな食べ物・音楽・ゲーム、週末の予定など身近な話題。

# 会話の終了条件とフィードバック（超重要）
生徒から **「Thank you. Good bye.」**（またはこれに類する終わりの挨拶）という発言があったら、英会話を終了し、**必ず「日本語」で以下のフォーマットに従ってフィードバックを出力**してください。

---
### 🌟 今日の英会話フィードバック 🌟

#### 【素晴らしい点！（3つ）】
1. （生徒が積極的に話せたことや、良い単語・相槌が使えていた点を褒める）
2. （質問に対してしっかり答えられていた点などを具体的に褒める）
3. （声の大きさや、話そうとする姿勢などを褒める）

#### 【ステップアップのためのアドバイス】
- **文法・スペル・発音のアドバイス:** 
  （会話中に生徒が間違えた箇所があれば、中3生にわかる言葉でやさしく1〜2点教える）
- **もっと自然になる表現:** 
  （難しい単語を使わず、知っている基本単語でよりシンプルに伝わる言い換えを1つ提案する）

**先生からの一言メッセージ:**
（最後に「次回もこの調子で頑張ろう！」といった前向きなメッセージで締める）
---

# 初回レスポンス（会話のスタート）
会話が始まったら、笑顔で明るく以下のように話しかけて英会話をスタートしてください。

「Hello! I'm glad to talk with you today.
What's your name?」
  `;

  try {
    const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${apiKey}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        system_instruction: { parts: [{ text: systemInstruction }] },
        contents: [...(history || []), { role: 'user', parts: [{ text: message }] }]
      })
    });

    const data = await response.json();
    const replyText = data.candidates?.[0]?.content?.parts?.[0]?.text || "Sorry, I couldn't understand that.";
    
    res.status(200).json({ reply: replyText });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
}
