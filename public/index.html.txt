<body>
  <h2>🎤 English Speaking Partner</h2>
  <div id="chat-box"></div>
  <button id="mic-btn" type="button">🎙️ 話す (Click & Speak)</button>

  <script>
    document.addEventListener('DOMContentLoaded', () => {
      const micBtn = document.getElementById('mic-btn');
      const chatBox = document.getElementById('chat-box');
      let history = [];

      const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;

      if (!SpeechRecognition) {
        alert('このブラウザは音声入力に対応していません。Google Chromeを使用してください。');
        micBtn.disabled = true;
        return;
      }

      const recognition = new SpeechRecognition();
      recognition.lang = 'en-US';

      micBtn.addEventListener('click', () => {
        try {
          recognition.start();
          micBtn.innerText = '👂 聞いています... (英語で話してください)';
          micBtn.disabled = true;
        } catch (e) {
          console.error(e);
        }
      });

      recognition.onresult = async (event) => {
        const userText = event.results[0][0].transcript;
        appendMessage('You', userText, 'user');
        micBtn.innerText = '🤔 考え中...';

        try {
          const res = await fetch('/api/chat', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ message: userText, history: history })
          });
          const data = await res.json();
          
          const reply = data.reply || data.error || "エラーが発生しました";
          appendMessage('AI', reply, 'ai');
          speak(reply);

          history.push({ role: 'user', parts: [{ text: userText }] });
          history.push({ role: 'model', parts: [{ text: reply }] });
        } catch (err) {
          appendMessage('AI', '通信エラーが発生しました。', 'ai');
        }

        micBtn.innerText = '🎙️ 話す (Click & Speak)';
        micBtn.disabled = false;
      };

      recognition.onerror = (event) => {
        alert('音声が聞き取れなかったか、マイクアクセスが拒否されました。');
        micBtn.innerText = '🎙️ 話す (Click & Speak)';
        micBtn.disabled = false;
      };

      function appendMessage(sender, text, className) {
        const div = document.createElement('div');
        div.className = className;
        div.innerText = `${sender}: ${text}`;
        chatBox.appendChild(div);
        chatBox.scrollTop = chatBox.scrollHeight;
      }

      function speak(text) {
        const synth = window.speechSynthesis;
        const utterance = new SpeechSynthesisUtterance(text);
        utterance.lang = 'en-US';
        synth.speak(utterance);
      }
    });
  </script>
